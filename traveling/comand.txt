from django.contrib.auth.models import User
from main.models import UserProfile, Notification, Trip
from datetime import datetime, timedelta

# Получаем всех пользователей
users = UserProfile.objects.all()

# Создаем различные типы уведомлений для каждого пользователя
for user in users:
    # 1. Информационное сообщение
    Notification.objects.create(
        recipient=user,
        message="Добро пожаловать в FastTravel! Мы рады видеть вас в нашем сервисе.",
        notification_type='info'
    )
    
    # 2. Предупреждение
    Notification.objects.create(
        recipient=user,
        message="Не забудьте подтвердить свой email для полного доступа к функциям сервиса.",
        notification_type='warning'
    )
    
    # 3. Успешное действие
    Notification.objects.create(
        recipient=user,
        message="Ваш профиль успешно обновлен.",
        notification_type='success'
    )
    
    # 4. Запрос на присоединение (если есть другие пользователи)
    other_users = UserProfile.objects.exclude(id=user.id)
    if other_users.exists():
        sender = other_users.first()
        # Находим или создаем тестовую поездку
        trip = Trip.objects.first()
        if trip:
            Notification.objects.create(
                recipient=user,
                sender=sender,
                trip=trip,
                message=f"{sender.first_name} {sender.last_name} хочет присоединиться к вашей поездке.",
                notification_type='request'
            )
    
    # 5. Уведомление о начале поездки
    if trip:
        Notification.objects.create(
            recipient=user,
            sender=user,  # В данном случае отправитель - сам пользователь
            trip=trip,
            message=f"Поездка из {trip.departure_city.name} в {trip.destination_city.name} началась!",
            notification_type='start'
        )
    
    # 6. Уведомление о завершении поездки
    if trip:
        Notification.objects.create(
            recipient=user,
            sender=user,
            trip=trip,
            message=f"Поездка из {trip.departure_city.name} в {trip.destination_city.name} завершена.",
            notification_type='end'
        )
    
    # 7. Уведомление об удалении из поездки
    if trip and other_users.exists():
        sender = other_users.first()
        Notification.objects.create(
            recipient=user,
            sender=sender,
            trip=trip,
            message=f"Вы были удалены из поездки {trip.departure_city.name} -> {trip.destination_city.name}.",
            notification_type='removed'
        )

print("Тестовые уведомления успешно созданы!")
